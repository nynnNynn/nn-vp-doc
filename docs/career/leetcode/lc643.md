# 643 子数组最大平均数 I
## code
```javascript
/**
 * @param {number[]} nums
 * @param {number} k
 * @return {number}
 */
var findMaxAverage = function(nums, k) {

    let max = 0;
    function getSum(left, right){
        let result = 0;
        for(let i = left; i < right; i++){
            result = result + nums[i];
        }
        return result;
    }

    let ind_left = 0; 
    let ind_right = k - 1;

    while (ind_right < k.length){
        max = Math.max(max, getSum(ind_left, ind_right));
        ind_right++;
        ind_left++;

    }
    return max / k;
};
```
