# 236 二叉树最近公共祖先
## code
```javascript
/**
 * Definition for a binary tree node.
 * function TreeNode(val) {
 *     this.val = val;
 *     this.left = this.right = null;
 * }
 */
/**
 * @param {TreeNode} root
 * @param {TreeNode} p
 * @param {TreeNode} q
 * @return {TreeNode}
 */
var lowestCommonAncestor = function(root, p, q) {
    let result;
    let temp = [];
    function traverse(root, temp_str){
        if(root === null){
            // console.log(temp_str);
            return;
        }
        if(root.val === p || root.val === q){
            temp.push(temp_str);
        }
        let temp = temp_str + root.val + ',';
        traverse(root.left, temp);
        traverse(root.right, temp);
    }
    
    traverse(root, '');
    console.log(temp);
    return result;
};
```
