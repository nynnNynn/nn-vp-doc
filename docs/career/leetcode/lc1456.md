# 1456 定长子串中元音的最大数目
## code
```javascript
/**
 * @param {string} s
 * @param {number} k
 * @return {number}
 */
var maxVowels = function(s, k) {

    let alpha = {
        'a': true,
        'e': true,
        'i': true,
        'u': true,
        'o': true,
    };
    let max = -Infinity;
    let temp = 0;
    for(let i = 0; i < s.length; i++){
        if(temp === k){
            return k;
        }
        // 辅音情况 
        if(!alpha[s.charAt(i)]){
            if(temp !== 0){
                max = Math.max(temp, max);
                temp = 0;
            }
            continue;
        }
        // 元音情况
        temp++;
    }
    return max;
};
```
