# 020 有效的括号
## code
```javascript
/**
 * @param {string} s
 * @return {boolean}
 */
var isValid = function(s) {
    let stack = [];
    const alpha = {
        '(':'）',
        '{':'}',
        '[':']',
    }
    for(let i = 0; i < s.length; i++){
      if(Object.keys(alpha).includes(s.charAt(i))){
        stack.push(s.charAt(i));
        continue;
      }
      if(s.charAt(i) !== alpha[stack[stack.length - 1]]){
        return false;

      }
      stack.pop();
       
      
    }
    return stack.length === 0;
    
};
```
