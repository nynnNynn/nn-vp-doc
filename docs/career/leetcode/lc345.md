var reverseVowels = function(s) {
    let ind_left = 0;
    let ind_right = s.length - 1;
    let vowels = [
        'a','i','e','o','u','A','I','E','O','U'
    ]
    while(ind_left < ind_right){
        console.log(ind_left);
        console.log(ind_right);

        if(!vowels.includes[s.charAt(ind_left)]){
            console.log('左辅音');
            console.log(s.charAt(ind_left));
            ind_left++;
        }
        if(!vowels.includes[s.charAt(ind_right)]){
            ind_right--;
        }
        continue;
    }
    
};
