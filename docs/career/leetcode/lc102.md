# 102 二叉树层序遍历
## code
```javascript
/**
 * Definition for a binary tree node.
 * function TreeNode(val, left, right) {
 *     this.val = (val===undefined ? 0 : val)
 *     this.left = (left===undefined ? null : left)
 *     this.right = (right===undefined ? null : right)
 * }
 */
/**
 * @param {TreeNode} root
 * @return {number[][]}
 */
var levelOrder = function(root) {
    let result = [];
    function traverse(root, floor){
        if(root === null){
            return;
        }
        if(result[floor] === undefined){
            result[floor] = [root.val];
        } else {
            result[floor].push(root.val);
        }

        traverse(root.left, (floor + 1));
        traverse(root.right, (floor + 1));
    }
    traverse(root, 0);
    return result;
};
```
