# 735 小行星碰撞
## code
```javascript
/**
 * @param {number[]} asteroids
 * @return {number[]}
 */
var asteroidCollision = function(asteroids) {
    let temp = [];
    for(let i = 0; i < asteroids.length; i++){

        if(asteroids[i] > 0){
            temp.push(asteroids[i]);
            continue;
            
        }
        if(asteroids[i] < 0){
            if(temp.length === 0){
                continue;
            }
            let init_length = temp.length;
            for(let j = init_length - 1; j >= 0; j--){
                if(Math.abs(temp[j]) > Math.abs(asteroids[i])){
                    break;
                }
                if(Math.abs(temp[j]) < Math.abs(asteroids[i])){
                    temp.pop();
                    if(temp.length === 0){
                        temp.push(asteroids[i]);
                    }
                }
                if(Math.abs(temp[j]) === Math.abs(asteroids[i])){
                    temp.pop();
                    break;
                }
            }
            
        }
    }
    return temp;
   
};
```
