## code
```javascript
/**
 * @param {number[]} height
 * @return {number}
 */
var maxArea = function(height) {
    let ind_left = 0;
    let ind_right = height.length - 1;
    let max = 0; 
    while(ind_left < ind_right){
        max = Math.max(max, Math.min(height[ind_left], height[ind_right]) * (ind_right - ind_left));
        if(height[ind_left] < height[ind_right]){
            ind_left++;
        } else {
            ind_right--;

        }
    }
    return max;
};
```
