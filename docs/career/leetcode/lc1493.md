# 1493删一个之后数组
## code
```javascript
/**
 * @param {number[]} nums
 * @return {number}
 */
var longestSubarray = function(nums) {
    let ind_left = 0;
    let ind_right = 0;
    let zero_count = 0;
    let max = 0;
    while(ind_right < nums.length){

        if(nums[ind_right] === 0){
            zero_count++;
        }

        while(zero_count > 1){
            if(nums[ind_left] === 0){
                zero_count--;
            }
            ind_left++;
        }
        max = Math.max(max, ind_right - ind_left);
        ind_right++;
    }
};
```
