# 2215 找出两数组的不同
## code
```javascript
/**
 * @param {number[]} nums1
 * @param {number[]} nums2
 * @return {number[][]}
 */
var findDifference = function(nums1, nums2) {
    let set1 = new Set(nums1);
    let set2 = new Set(nums2);
    let result1 = [];
    let result2 = [];
    set1.forEach(
        (item) => {
            if(!set2.has(item)){
                result1.push(item);
            }
        }
    );
    set2.forEach(
        (item) => {
            if(!set1.has(item)){
                result2.push(item);
            }
        }
    );
    return [result1, result2]; 
};
```
