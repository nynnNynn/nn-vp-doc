/**
 * @param {number[][]} rooms
 * @return {boolean}
 */
var canVisitAllRooms = function(rooms) {
    let unlocked = new Array(rooms.length).fill(false);
    function traverse(index){
        if(unlocked[index]){
            return;
        }
        unlocked[index] = true;
        rooms[index].forEach(
            (item) => {
                traverse(item)
            }
        );
    }
    traverse(0);
    return unlocked.every(
        (item) => {return item === true}
    );  
};
