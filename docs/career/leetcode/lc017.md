# 017 电话号码的字母组合
## code
```javascript
/**
 * @param {string} digits
 * @return {string[]}
 */
var letterCombinations = function(digits) {
    let result = [];
    const alpha_map = { 
        2: ['a','b','c'],
        3: ['d','e','f'],
        4: ['g','h','i'],
        5: ['j','k','l'],
        6: ['m','n','o'],
        7: ['p','q','r','s'],
        8: ['t','u','v'],
        9: ['w','x','y','z'],
    }
    function backtrace(temp, ind){
        if(temp.length === digits.length){
            result.push(temp);
            return;
        }
        let digit = alpha_map[digits.charAt(Number(ind))];
    
        for(let i = 0; i < digit.length; i++){
          
            backtrace(temp + digit[i], (ind + 1));
        }
    }
    backtrace('', 0);
    return result;
};
```
