var canFinish = function(numCourses, prerequisites) {
    // ab ba 锁死情况是不行的
    // ab b只要没有的情况就是不行的就可以修
    // ab bc ca 行么 也不行 的
    let head = null;
    let created_map = {
    };
    prerequisites.forEach(
        (item) => {
            
            if(created_map[item[0]] === undefined){
                let root1 = {
                    val: item[0],
                }
                created_map[item[0]] = root1;
                if(Object.keys(created_map).length === 1){
                    head = created_map[item[0]];
                }
                

            } 
            if(created_map[item[1]] === undefined){
                let root2 = {
                    val: item[1],
                }
                created_map[item[1]] = root2;
            }
            created_map[item[0]].next = created_map[item[1]];
        }
    );
    let ind_slow = head;
    let ind_fast = head;
    if(ind_fast.next === null || ind_fast.next.next === null){
        return true;
    }
    ind_slow = ind_slow.next;
    ind_fast = ind_fast.next.next;
    while(ind_fast !== null && ind_fast.next !== null){
        if(ind_fast === ind_slow){
            console.log('这');
            return false;
        }
        ind_fast = ind_fast.next;
        ind_fast = ind_fast.next;
        ind_slow = ind_slow.next;
    }
    
    return true;
};
