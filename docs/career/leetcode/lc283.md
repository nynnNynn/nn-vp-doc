/**
 * @param {number[]} nums
 * @return {void} Do not return anything, modify nums in-place instead.
 */
var moveZeroes = function(nums) {
    let ind = 0;
    let position;
    while(ind !== nums.length - 1){

        if(nums[ind] === 0){
            position = ind;
            ind++;
        } else {
            if(position !== undefined){
                [nums[ind], nums[position]] = [nums[position], nums[ind]];
                position = undefined;
            }
            
        }
        console.log('看看每次');
        console.log(ind);
        console.log(position);
    }

    
};
