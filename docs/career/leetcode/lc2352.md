# 2352 相等行列对
## code
```javascript
/**
 * @param {number[][]} grid
 * @return {number}
 */
var equalPairs = function(grid) {
    let rows = grid.map((item) => {
        return item.join(',');
    });
    let map_row = {};
    rows.forEach(
        (item) => {
            if(map_row[item] === undefined){
                map_row[item] = 1;
            } else {
                map_row[item]++;
            }
        }
    );
    let count = 0;
    for(let i = 0; i < grid.length; i++){
        let temp = ''
        grid.forEach(
            (item, id) => {
                if(id === 0){
                    temp = temp + grid[id][i];
                } else {
                    temp = temp + ',' + grid[id][i];
                }
            }
        );
        if(map_row[temp]){
            count = count + map_row[temp];
        }
    }

    return count;
};
```
