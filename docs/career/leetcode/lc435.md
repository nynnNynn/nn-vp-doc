# 435 无重叠区间
## code
```javascript
// 貌似是可以贪心，两个遍历，一旦 第二个左区间 小于 前一个右区间，就给他砍了
/**
 * @param {number[][]} intervals
 * @return {number}
 */
var eraseOverlapIntervals = function(intervals) {

    intervals.sort((a, b) => a[1] - b[1]);
    let count = 0;
    let right_side = intervals[0][1];
    for(let i = 1; i < intervals.length; i++){

        if(intervals[i][0] < right_side){
            count++;
            right_side = Math.min(right_side, intervals[i][1]);
            continue;
        }
        right_side = intervals[i][1];

    }
    return count;
};
```
