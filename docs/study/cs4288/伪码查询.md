## HTML fetch 拉接口然后挂dom操作
拉接口返回这种东西，找dom id去挂
```js
{
  "commands": [
    { "type": "ADD", "id": "a", "value": "Apple" },
    { "type": "ADD", "id": "b", "value": "Banana" },
    { "type": "UPDATE", "id": "a", "value": "Orange" },
    { "type": "DELETE", "id": "b" }
  ]
}
```
写法：
```js

<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>DOM Operations</title>
</head>
<body>
  <!-- 题目给什么 ID，就把对应 ID 填到这里 -->
  <button id="loadButton">Load</button>
  <ul id="itemList"></ul>
  <p id="statusDisplay">Ready</p>

  <script>
    // 1. 找到题目指定的元素
    // 要监听谁，就获取谁的 ID
    const button = document.getElementById("loadButton");

    // 要添加、更新、删除哪里的内容，就获取哪个容器
    const list = document.getElementById("itemList");

    // 要在哪里显示状态、结果或错误，就获取哪个元素
    const status = document.getElementById("statusDisplay");

    // 2. 统一处理 DOM 的添加、修改、删除
    function applyCommand(command) {
      // command.id 是服务端返回的目标 ID
      // 用 data-id 查找对应的 li
      const item = Array.from(list.children).find(
        child => child.dataset.id === String(command.id)
      );

      // 题目说 ADD / CREATE / INSERT，就进添加分支
      if (command.type === "ADD") {
        // 已存在就不重复添加；题目另有要求可修改
        if (item) return;

        // 要创建什么标签，就把 li 改为 div、p、span 等
        const newItem = document.createElement("li");

        // 保存业务 ID，方便后续查找
        newItem.dataset.id = command.id;

        // 题目要求显示哪个字段，就取哪个字段
        newItem.textContent = command.value;

        // 题目要求添加到哪个容器，就在那个容器 append
        list.appendChild(newItem);

      // 题目说 UPDATE / MODIFY / CHANGE，就进更新分支
      } else if (command.type === "UPDATE") {
        if (item) {
          // 修改文本用 textContent
          // 修改输入框的值则改成 item.value = command.value
          item.textContent = command.value;
        }

      // 题目说 DELETE / REMOVE，就进删除分支
      } else if (command.type === "DELETE") {
        if (item) {
          // 删除找到的整个元素
          item.remove();
        }
      }
    }

    // 3. 绑定事件
    // 题目要 click 就填 click，其他如 change、submit 同理
    button.addEventListener("click", async () => {
      // 请求开始时更新提示
      status.textContent = "Loading...";

      try {
        // 4. 题目给哪个 API，就替换这里的 URL
        const response = await fetch("/api/items");

        // 5. HTTP 404/500 不会让 fetch 自动 reject
        if (!response.ok) {
          throw new Error("HTTP " + response.status);
        }

        // 6. JSON 需要再 await 一次
        const data = await response.json();

        // 7. 题目返回数组在哪个字段，就换 data.commands
        // 比如 data.items、data.results
        for (const command of data.commands) {
          applyCommand(command);
        }

        // 8. 题目要求显示什么最终结果，就在这里写
        status.textContent = list.children.length + " items";

      } catch (error) {
        // 网络错误、JSON 解析错误以及主动 throw 都会到这里
        status.textContent = "Error: " + error.message;
      }
    });
  </script>
</body>
</html>

```
## express app，中间件和mongodb
```js
const express = require("express");
const { MongoClient } = require("mongodb");

const app = express();
const router = express.Router();

app.use(express.json());

// ======================
// 1. MongoDB
// ======================

const client = new MongoClient("mongodb://localhost:27017");

// ======================
// 2. 三个中间件
// ======================

// 日志中间件
function logger(req, res, next) {
    console.log(req.method, req.path);
    next();
}

// 鉴权中间件
function auth(req, res, next) {
    // 假设校验 token
    next();
}

// 风控中间件
function riskCheck(req, res, next) {
    // 假设执行风控检查
    next();
}

// ======================
// 3. 只对 /user 生效
// ======================

router.use(logger);
router.use(auth);
router.use(riskCheck);

// ======================
// 4. CRUD
// ======================

async function main() {
    await client.connect();

    const db = client.db("school");
    const users = db.collection("users");

    // GET /user/hemingway
    router.get("/:name", async (req, res) => {
        try {
            const user = await users.findOne({
                name: req.params.name
            });

            if (!user) {
                return res.sendStatus(404);
            }

            res.json(user);
        } catch (err) {
            res.sendStatus(500);
        }
    });

    // GET /user
    router.get("/", async (req, res) => {
        try {
            const data = await users.find({}).toArray();
            res.json(data);
        } catch (err) {
            res.sendStatus(500);
        }
    });

    // POST /user
    router.post("/", async (req, res) => {
        try {
            const result = await users.insertOne(req.body);
            res.status(201).json(result);
        } catch (err) {
            res.sendStatus(500);
        }
    });

    // PUT /user/hemingway
    router.put("/:name", async (req, res) => {
        try {
            const result = await users.updateOne(
                { name: req.params.name },
                { $set: req.body }
            );

            if (result.matchedCount === 0) {
                return res.sendStatus(404);
            }

            res.json(result);
        } catch (err) {
            res.sendStatus(500);
        }
    });

    // DELETE /user/hemingway
    router.delete("/:name", async (req, res) => {
        try {
            const result = await users.deleteOne({
                name: req.params.name
            });

            if (result.deletedCount === 0) {
                return res.sendStatus(404);
            }

            res.sendStatus(204);
        } catch (err) {
            res.sendStatus(500);
        }
    });

    // 把整个 router 挂到 /user
    app.use("/user", router);

    app.listen(3000, () => {
        console.log("Server running on port 3000");
    });
}

main().catch(console.error);
```

```js
function writeLog(name) {
    console.log("Logger:", name);
}

function checkAuth(user) {
    console.log("Auth:", user);
    return true;
}

function checkRisk(user) {
    console.log("Risk:", user);
    return true;
}

router.get("/:name", (req, res) => {
    // 业务 1
    console.log("Business 1");

    // 手动调用日志函数
    writeLog(req.params.name);

    // 业务 2、3
    console.log("Business 2");
    console.log("Business 3");

    // 手动鉴权与风控
    if (!checkAuth(req.user)) {
        return res.sendStatus(401);
    }

    if (!checkRisk(req.user)) {
        return res.sendStatus(403);
    }

    // 业务 4、5
    console.log("Business 4");
    console.log("Business 5");

    res.send("Done");
});
```
