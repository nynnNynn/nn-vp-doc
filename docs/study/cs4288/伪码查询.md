## HTML fetch 拉接口然后挂dom操作
拉接口返回这种东西，找dom id去挂
```js
{
  "commands": [
    { "type": "ADD", "id": "a", "value": "Apple" },
    { "type": "ADD", "id": "b", "value": "Banana" },
    { "type": "UPDATE", "id": "a", "value": "Orange" },
    { "type": "DELETE", "id": "b" }
  ]
}
```
写法：
```js

<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>DOM Operations</title>
</head>
<body>
  <!-- 题目给什么 ID，就把对应 ID 填到这里 -->
  <button id="loadButton">Load</button>
  <ul id="itemList"></ul>
  <p id="statusDisplay">Ready</p>

  <script>
    // 1. 找到题目指定的元素
    // 要监听谁，就获取谁的 ID
    const button = document.getElementById("loadButton");

    // 要添加、更新、删除哪里的内容，就获取哪个容器
    const list = document.getElementById("itemList");

    // 要在哪里显示状态、结果或错误，就获取哪个元素
    const status = document.getElementById("statusDisplay");

    // 2. 统一处理 DOM 的添加、修改、删除
    function applyCommand(command) {
      // command.id 是服务端返回的目标 ID
      // 用 data-id 查找对应的 li
      const item = Array.from(list.children).find(
        child => child.dataset.id === String(command.id)
      );

      // 题目说 ADD / CREATE / INSERT，就进添加分支
      if (command.type === "ADD") {
        // 已存在就不重复添加；题目另有要求可修改
        if (item) return;

        // 要创建什么标签，就把 li 改为 div、p、span 等
        const newItem = document.createElement("li");

        // 保存业务 ID，方便后续查找
        newItem.dataset.id = command.id;

        // 题目要求显示哪个字段，就取哪个字段
        newItem.textContent = command.value;

        // 题目要求添加到哪个容器，就在那个容器 append
        list.appendChild(newItem);

      // 题目说 UPDATE / MODIFY / CHANGE，就进更新分支
      } else if (command.type === "UPDATE") {
        if (item) {
          // 修改文本用 textContent
          // 修改输入框的值则改成 item.value = command.value
          item.textContent = command.value;
        }

      // 题目说 DELETE / REMOVE，就进删除分支
      } else if (command.type === "DELETE") {
        if (item) {
          // 删除找到的整个元素
          item.remove();
        }
      }
    }

    // 3. 绑定事件
    // 题目要 click 就填 click，其他如 change、submit 同理
    button.addEventListener("click", async () => {
      // 请求开始时更新提示
      status.textContent = "Loading...";

      try {
        // 4. 题目给哪个 API，就替换这里的 URL
        const response = await fetch("/api/items");

        // 5. HTTP 404/500 不会让 fetch 自动 reject
        if (!response.ok) {
          throw new Error("HTTP " + response.status);
        }

        // 6. JSON 需要再 await 一次
        const data = await response.json();

        // 7. 题目返回数组在哪个字段，就换 data.commands
        // 比如 data.items、data.results
        for (const command of data.commands) {
          applyCommand(command);
        }

        // 8. 题目要求显示什么最终结果，就在这里写
        status.textContent = list.children.length + " items";

      } catch (error) {
        // 网络错误、JSON 解析错误以及主动 throw 都会到这里
        status.textContent = "Error: " + error.message;
      }
    });
  </script>
</body>
</html>
```
## 浏览器post表单
```js

<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <title>Task Form</title>
</head>
<body>
  <form id="taskForm">
    <input name="title" required>

    <select name="priority">
      <option value="low">Low</option>
      <option value="high">High</option>
    </select>

    <button type="submit">Save</button>
  </form>

  <p id="message"></p>
  <ul id="taskList"></ul>

  <script>
    // 1. 题目给哪个 form ID，就替换这里
    const form = document.getElementById("taskForm");
    const message = document.getElementById("message");
    const list = document.getElementById("taskList");

    // 2. 监听 submit，而不是只监听按钮 click
    form.addEventListener("submit", async (event) => {
      // 取消原生提交和整页刷新
      event.preventDefault();

      // 3. 根据 input 的 name 读取字段
      // 题目给 name="email" 就用 data.get("email")
      const data = new FormData(form);

      const title = String(data.get("title") ?? "").trim();
      const priority = data.get("priority");

      // 4. 题目有什么校验要求，就在这里检查
      if (!title) {
        message.textContent = "Title required";
        return;
      }

      try {
        // 5. 题目给哪个 API，就改 URL
        // 创建通常用 POST
        // 更新可用 PUT/PATCH
        // 删除可用 DELETE
        const response = await fetch("/api/tasks", {
          method: "POST",

          // 发送 JSON 时指定 Content-Type
          headers: {
            "Content-Type": "application/json"
          },

          // 题目要求提交哪些字段，就放进对象
          body: JSON.stringify({ title, priority })
        });

        if (!response.ok) {
          throw new Error("HTTP " + response.status);
        }

        // 假设成功响应包含创建后的 JSON 对象
        const task = await response.json();

        // 6. 要显示什么元素，就创建什么标签
        const li = document.createElement("li");

        // 题目返回哪个字段就读取哪个字段
        li.textContent = task.title;

        list.appendChild(li);

        // 7. 成功后清空表单
        form.reset();
        message.textContent = "Saved";

      } catch (error) {
        message.textContent = error.message;
      }
    });
  </script>
</body>
</html>
```

## express router， app和中间件
```js
const express = require("express");

const app = express();
const router = express.Router();

// 1. 解析 JSON 请求体
// 否则 POST/PUT 请求的 req.body 可能是 undefined
app.use(express.json());

// 2. 假数据库
// 题目给初始数据，就替换这个数组
let books = [
  { id: 1, title: "Book A", price: 10 },
  { id: 2, title: "Book B", price: 20 }
];

// 用于生成新 ID
let nextId = 3;


// ===== Middleware =====
// 中间件当函数用，next感觉可以理解成控制流，看app.use里面的注册顺序，他要走那个基本的一次流程单位，
function logger(req, res, next) {
  console.log(req.method, req.originalUrl);
  next();
}

// 鉴权中间件
function auth(req, res, next) {
  // 题目说从哪个 Header 读取密钥，就改这里
  const key = req.headers["x-api-key"];

  // 题目规定什么认证条件，就替换这个判断
  if (key !== "secret123") {
    return res.status(401).json({
      error: "Unauthorized"
    });
  }

  // 认证成功继续
  next();
}

// 验证请求体
function validateBook(req, res, next) {
  // 题目要求哪些字段，就在这里检查哪些字段
  const { title, price } = req.body;

  if (
    typeof title !== "string" ||
    title.trim() === "" ||
    typeof price !== "number" ||
    price < 0
  ) {
    return res.status(400).json({
      error: "Invalid book data"
    });
  }

  next();
}

// 3. 所有 /api/books 下的请求都经过这两个中间件
// 注册顺序：logger -> auth -> 对应路由
// 匹配所有路径
router.use(logger);
app.use("/user", auth);   // 仅匹配 /user 路径前缀
router.use(auth);


// ===== GET ALL =====

// 题目要求获取整个集合，就使用 GET /
router.get("/", (req, res) => {
  // 可选 query 过滤
  // 如 GET /api/books?minPrice=15
  const { minPrice } = req.query;

  let result = books;

  if (minPrice !== undefined) {
    const min = Number(minPrice);

    if (Number.isNaN(min)) {
      return res.status(400).json({
        error: "Invalid minPrice"
      });
    }

    result = result.filter(book => book.price >= min);
  }

  res.status(200).json(result);
});


// ===== GET ONE =====

// :id 代表动态路径参数
// GET /api/books/1 -> req.params.id = "1"
router.get("/:id", (req, res) => {
  // URL 参数是字符串，数字 ID 需要转换
  const id = Number(req.params.id);

  // 如果题目用 username 作为 ID，就不需要 Number
  const book = books.find(book => book.id === id);

  if (!book) {
    return res.status(404).json({
      error: "Not found"
    });
  }

  res.status(200).json(book);
});


// ===== CREATE =====

// POST 通常用于创建
// validateBook 只注册在这个路由上
router.post("/", validateBook, (req, res) => {
  // 题目要求哪些字段，就从 req.body 提取
  const { title, price } = req.body;

  const newBook = {
    id: nextId++,
    title,
    price
  };

  books.push(newBook);

  // 创建成功通常 201
  res.status(201).json(newBook);
});


// ===== UPDATE =====

// PUT 更新指定 ID 对应的对象
router.put("/:id", validateBook, (req, res) => {
  const id = Number(req.params.id);

  const book = books.find(book => book.id === id);

  if (!book) {
    return res.status(404).json({
      error: "Not found"
    });
  }

  // 题目要求更新哪些字段，就赋值哪些字段
  book.title = req.body.title;
  book.price = req.body.price;

  res.status(200).json(book);
});


// ===== DELETE =====

router.delete("/:id", (req, res) => {
  const id = Number(req.params.id);

  const index = books.findIndex(book => book.id === id);

  if (index === -1) {
    return res.status(404).json({
      error: "Not found"
    });
  }

  // 删除数组里对应位置的一个元素
  books.splice(index, 1);

  // 删除成功，无需返回 JSON 时用 204
  res.status(204).send();
});


// ===== Mount Router =====

// router 内的 "/" 对应 /api/books
// router 内的 "/:id" 对应 /api/books/:id
// 注册出来一个完整的业务服务上的域吧
app.use("/api/books", router);


// ===== Error Handling =====

// 没有任何路由匹配时返回 404
app.use((req, res) => {
  res.status(404).json({
    error: "Route not found"
  });
});

// 错误处理中间件有四个参数，err 放第一个
app.use((err, req, res, next) => {
  console.error(err);

  res.status(500).json({
    error: "Internal server error"
  });
});


// ===== Start Server =====

// 题目给哪个端口，就修改这里
app.listen(3000, () => {
  console.log("Server running on port 3000");
});
```

## mongoose 函数
打这么个接口 GET /api/products?category=food&sort=asc
```js

const express = require("express");
const mongoose = require("mongoose");

const app = express();
app.use(express.json());


// ===== 1. Database Connection =====

// 题目给哪个 MongoDB URL，就替换这里
const DB_URL = "mongodb://127.0.0.1:27017/shop";


// ===== 2. Schema =====

// 题目描述哪些字段，就在这里定义哪些字段
const productSchema = new mongoose.Schema({
  name: {
    type: String,
    required: true
  },

  price: {
    type: Number,
    required: true,
    min: 0
  },

  category: {
    type: String,
    default: "general"
  },

  inStock: {
    type: Boolean,
    default: true
  }
});

// 创建 Model
// 第一个参数是 Model 名称
const Product = mongoose.model("Product", productSchema);


// ===== 3. CREATE =====

app.post("/api/products", async (req, res) => {
  try {
    // 题目给什么字段，就在这里取什么字段
    const { name, price, category, inStock } = req.body;

    // 用 Model 创建一条 MongoDB 文档
    const product = await Product.create({
      name,
      price,
      category,
      inStock
    });

    res.status(201).json(product);

  } catch (error) {
    // Mongoose Schema 校验失败
    if (error.name === "ValidationError") {
      return res.status(400).json({
        error: error.message
      });
    }

    res.status(500).json({ error: error.message });
  }
});


// ===== 4. READ ALL + FILTER + SORT =====

app.get("/api/products", async (req, res) => {
  try {
    // URL ?category=food -> req.query.category
    // URL ?sort=desc -> req.query.sort
    const { category, sort } = req.query;

    // 题目要求按什么条件筛选，就把条件放进 filter
    const filter = {};

    if (category) {
      filter.category = category;
    }

    // 1 升序，-1 降序
    const sortOrder = sort === "desc" ? -1 : 1;

    const products = await Product.find(filter)
      .sort({ price: sortOrder });

    res.status(200).json(products);

  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});


// ===== 5. READ ONE =====

app.get("/api/products/:id", async (req, res) => {
  try {
    // MongoDB 文档默认使用 _id
    // 这里的 id 是 URL 中的 ID
    const product = await Product.findById(req.params.id);

    if (!product) {
      return res.status(404).json({
        error: "Not found"
      });
    }

    res.json(product);

  } catch (error) {
    // 无效 ObjectId 等转换错误
    if (error.name === "CastError") {
      return res.status(400).json({
        error: "Invalid ID"
      });
    }

    res.status(500).json({ error: error.message });
  }
});


// ===== 6. UPDATE =====

app.put("/api/products/:id", async (req, res) => {
  try {
    // 按题目要求更新允许修改的字段
    const { name, price, category, inStock } = req.body;

    // 仅放入实际提供的字段
    const updates = {};

    if (name !== undefined) updates.name = name;
    if (price !== undefined) updates.price = price;
    if (category !== undefined) updates.category = category;
    if (inStock !== undefined) updates.inStock = inStock;

    // 题目若要求完整替换，应另行校验全部必填字段
    // 本示例实现的是按提供字段更新
    const product = await Product.findByIdAndUpdate(
      req.params.id,
      { $set: updates },
      {
        new: true,           // 返回更新后的对象
        runValidators: true // 执行 Schema 更新校验
      }
    );

    if (!product) {
      return res.status(404).json({
        error: "Not found"
      });
    }

    res.status(200).json(product);

  } catch (error) {
    if (["CastError", "ValidationError"].includes(error.name)) {
      return res.status(400).json({ error: error.message });
    }

    res.status(500).json({ error: error.message });
  }
});


// ===== 7. DELETE =====

app.delete("/api/products/:id", async (req, res) => {
  try {
    const product = await Product.findByIdAndDelete(
      req.params.id
    );

    if (!product) {
      return res.status(404).json({
        error: "Not found"
      });
    }

    res.status(204).send();

  } catch (error) {
    if (error.name === "CastError") {
      return res.status(400).json({ error: "Invalid ID" });
    }

    res.status(500).json({ error: error.message });
  }
});


// ===== 8. Start =====

async function main() {
  try {
    // 必须先连接数据库，再开始提供服务
    await mongoose.connect(DB_URL);

    app.listen(3000, () => {
      console.log("Server running on port 3000");
    });

  } catch (error) {
    console.error("Database connection failed:", error);
    process.exitCode = 1;
  }
}

main();
```

## 一个react组件示范例子
```js

import { useState, useEffect } from "react";

export default function TaskManager({ apiUrl = "/api/tasks" }) {


  const [tasks, setTasks] = useState([]);
  const [title, setTitle] = useState("");
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    // 用于组件卸载或 apiUrl 改变时取消请求
    const controller = new AbortController();

    async function loadData() {
      setLoading(true);
      setError("");

      try {
        // 题目要求请求什么地址，就使用 apiUrl
        const response = await fetch(apiUrl, {
          signal: controller.signal
        });

        if (!response.ok) {
          throw new Error("HTTP " + response.status);
        }

        const data = await response.json();

        // 本题假设 API 直接返回数组
        // 若返回 { tasks: [...] }，改为 setTasks(data.tasks)
        setTasks(data);

      } catch (err) {
        // 组件卸载时取消请求，不显示取消错误
        if (err.name !== "AbortError") {
          setError(err.message);
        }

      } finally {
        if (!controller.signal.aborted) {
          setLoading(false);
        }
      }
    }

    loadData();

    return () => controller.abort();

  // 题目要求依赖哪个 props，就放进依赖数组
  }, [apiUrl]);



  async function addTask() {
    if (!title.trim()) return;

    try {
      // 创建任务 -> POST
      const response = await fetch(apiUrl, {
        method: "POST",
        headers: {
          "Content-Type": "application/json"
        },

        // 题目要求传什么字段，就写进 JSON
        body: JSON.stringify({
          title,
          completed: false
        })
      });

      if (!response.ok) {
        throw new Error("Cannot create task");
      }

      const newTask = await response.json();

      // 添加到原数组末尾
      // React 中不要直接 tasks.push(newTask)
      setTasks(prev => [...prev, newTask]);

      // 清空输入框
      setTitle("");
      setError("");

    } catch (err) {
      setError(err.message);
    }
  }

  async function toggleTask(task) {
    // 题目要求修改哪个字段，就在这里计算新值
    const completed = !task.completed;

    try {
      // 更新指定 ID -> PATCH
      const response = await fetch(`${apiUrl}/${task.id}`, {
        method: "PATCH",
        headers: {
          "Content-Type": "application/json"
        },
        body: JSON.stringify({ completed })
      });

      if (!response.ok) {
        throw new Error("Cannot update task");
      }

      // 假设 API 返回更新后的完整任务
      const updatedTask = await response.json();

      // 对应 ID 的元素替换，其余保持不变
      setTasks(prev =>
        prev.map(item =>
          item.id === task.id ? updatedTask : item
        )
      );

      setError("");

    } catch (err) {
      setError(err.message);
    }
  }


  // ===== Delete =====

  async function deleteTask(id) {
    try {
      const response = await fetch(`${apiUrl}/${id}`, {
        method: "DELETE"
      });

      if (!response.ok) {
        throw new Error("Cannot delete task");
      }

      // 把对应 ID 从数组过滤掉
      setTasks(prev =>
        prev.filter(task => task.id !== id)
      );

      setError("");

    } catch (err) {
      setError(err.message);
    }
  }


  // ===== JSX Output =====

  return (
    <div>
      <h2>Task Manager</h2>

      {/* 题目要求显示 loading，就用条件渲染 */}
      {loading && <p>Loading...</p>}

      {/* 有错误时才显示 */}
      {error && <p role="alert">{error}</p>}

      {/* 受控输入：value 和 onChange 成对出现 */}
      <input
        value={title}
        onChange={event => setTitle(event.target.value)}
        placeholder="Task title"
      />

      {/* 点击执行添加函数 */}
      <button onClick={addTask}>
        Add
      </button>

      {/* 数据数组用 map 转换为 JSX */}
      <ul>
        {tasks.map(task => (
          <li key={task.id}>

            {/* 显示题目指定的字段 */}
            <span>{task.title}</span>

            {/* 条件渲染状态 */}
            <span>
              {task.completed ? "Done" : "Pending"}
            </span>

            {/* 传当前任务给处理函数 */}
            <button onClick={() => toggleTask(task)}>
              Toggle
            </button>

            {/* 传当前任务 ID 给删除函数 */}
            <button onClick={() => deleteTask(task.id)}>
              Delete
            </button>

          </li>
        ))}
      </ul>
    </div>
  );
}
```

```js
const express = require("express");
const session = require("express-session");

const app = express();

// 1. 解析 JSON body
app.use(express.json());


// ===== Session Configuration =====

// 必须在需要 session 的路由之前注册
app.use(session({
  // Cookie 的名称，题目给什么就填什么
  name: "sessionId",

  // 实际部署应从环境变量读取 secret
  // 这里用固定字符串方便考试代码展示
  secret: "exam-secret-change-in-production",

  resave: false,
  saveUninitialized: false,

  cookie: {
    httpOnly: true,
    secure: false,  // 本地 HTTP 演示；HTTPS 部署应使用 true
    sameSite: "lax"
  }
}));


// ===== Demo Users =====

// 假设题目给出了验证账号密码的方法
// 这里用假数据代替数据库
// 正式项目必须使用安全的密码哈希验证
const users = [
  {
    id: 1,
    username: "alice",
    password: "123456"
  }
];


// ===== Login =====

app.post("/login", (req, res, next) => {
  // 题目使用什么字段，就替换 username/password
  const { username, password } = req.body;

  // 题目说如何验证，就修改这个逻辑
  const user = users.find(
    u => u.username === username &&
         u.password === password
  );

  if (!user) {
    return res.status(401).json({
      error: "Invalid credentials"
    });
  }

  // 登录成功，更新 Session ID
  req.session.regenerate(err => {
    if (err) return next(err);

    // 把验证过的身份写入 session
    // req.session.userId 是自定义字段
    req.session.userId = user.id;
    req.session.username = user.username;

    req.session.save(err => {
      if (err) return next(err);

      res.json({ message: "Logged in" });
    });
  });
});


// ===== Authentication Middleware =====

function requireAuth(req, res, next) {
  // 是否存在有效登录信息
  if (!req.session.userId) {
    return res.status(401).json({
      error: "Please log in"
    });
  }

  // 有登录信息，放行到下一个处理器
  next();
}


// ===== Protected Route =====

// 需要保护哪个路由，就把 requireAuth 放到哪个路由上
app.get("/profile", requireAuth, (req, res) => {
  // 从 session 读取已经验证的用户身份
  res.json({
    id: req.session.userId,
    username: req.session.username
  });
});


// ===== Logout =====

app.post("/logout", (req, res, next) => {
  // 销毁服务端 session
  req.session.destroy(err => {
    if (err) return next(err);

    // 清除浏览器对应的 session Cookie
    res.clearCookie("sessionId", { path: "/" });

    res.json({ message: "Logged out" });
  });
});


// ===== Error Handler =====

app.use((err, req, res, next) => {
  console.error(err);

  res.status(500).json({
    error: "Internal server error"
  });
});


// ===== Start =====

app.listen(3000, () => {
  console.log("Server running on port 3000");
});
```
